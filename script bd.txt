-- Crear tabla citas_anuladas para registrar citas anuladas
CREATE TABLE citas_anuladas (
    id INT PRIMARY KEY IDENTITY(1,1),
    id_cita INT,
    fecha_anulacion DATE,
    comentario NVARCHAR(255),
    FOREIGN KEY (id_cita) REFERENCES citas(id)
);

--crear tabla citas modificadas
CREATE TABLE citas_modificadas (
    id INT PRIMARY KEY IDENTITY(1,1),
    id_cita INT,
    fecha_modificacion DATE,
    comentario NVARCHAR(255),
    FOREIGN KEY (id_cita) REFERENCES citas(id)
);

-- Procedimiento almacenado para insertar una nueva cita
CREATE PROCEDURE InsertarCita
    @nombre NVARCHAR(100),
    @apellido NVARCHAR(100),
    @telefono NVARCHAR(20),
    @correo VARCHAR(100),
    @direccion NVARCHAR(255),
    @domicilio BIT,
    @fecha DATE,
    @hora TIME,
    @id_pago INT,
    @id_empresa INT,
    @numero_ticket VARCHAR(50),
    @cantidad_productos INT
AS
BEGIN
    INSERT INTO citas (nombre, apellido, telefono, correo, direccion, domicilio, fecha, hora, id_pago, id_empresa, numero_ticket, cantidad_productos)
    VALUES (@nombre, @apellido, @telefono, @correo, @direccion, @domicilio, @fecha, @hora, @id_pago, @id_empresa, @numero_ticket, @cantidad_productos);
END;
-- Procedimiento almacenado para modificar una cita existente
CREATE PROCEDURE ModificarCita
    @id INT,
    @nombre NVARCHAR(100),
    @apellido NVARCHAR(100),
    @telefono NVARCHAR(20),
    @correo VARCHAR(100),
    @direccion NVARCHAR(255),
    @domicilio BIT,
    @fecha DATE,
    @hora TIME,
    @id_pago INT,
    @id_empresa INT,
    @numero_ticket VARCHAR(50),
    @cantidad_productos INT
AS
BEGIN
    UPDATE citas
    SET nombre = @nombre,
        apellido = @apellido,
        telefono = @telefono,
        correo = @correo,
        direccion = @direccion,
        domicilio = @domicilio,
        fecha = @fecha,
        hora = @hora,
        id_pago = @id_pago,
        id_empresa = @id_empresa,
        numero_ticket = @numero_ticket,
        cantidad_productos = @cantidad_productos
    WHERE id = @id;
END;

-- Procedimiento almacenado para anular una cita
CREATE PROCEDURE AnularCita
    @id INT,
    @comentario NVARCHAR(255)
AS
BEGIN
    -- Insertar en la tabla citas_anuladas
    INSERT INTO citas_anuladas (id_cita, fecha_anulacion, comentario)
    VALUES (@id, GETDATE(), @comentario);

    -- Eliminar la cita de la tabla citas
    DELETE FROM citas WHERE id = @id;
END;
-- Procedimiento almacenado para consultar citas modificadas y anuladas
CREATE OR ALTER PROCEDURE ConsultarCitasModificadasYAnuladas
AS
BEGIN
    -- Citas que están modificadas (aquí podrías llevar un control adicional si lo necesitas)
    SELECT 
        c.id, c.nombre, c.apellido, c.telefono, c.fecha, c.hora, 
        'Modificada' AS estado
    FROM citas c
    WHERE c.id IN (SELECT id_cita FROM citas_anuladas) -- Suponiendo que luego restauras una cita o la marcas como modificada

    UNION

    -- Citas anuladas (se consultan desde la tabla citas_anuladas y se une con citas para obtener los datos)
    SELECT 
        c.id, c.nombre, c.apellido, c.telefono, c.fecha, c.hora, 
        'Anulada' AS estado
    FROM citas_anuladas ca
    INNER JOIN citas c ON ca.id_cita = c.id;
END;
